cmake_minimum_required(VERSION 3.25)
project(StellaVibra VERSION 0.2.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Windows x64 and MSVC are required")
endif()
enable_language(RC)
configure_file(src/version.rc.in "${CMAKE_BINARY_DIR}/version.rc" @ONLY)
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Widgets Network)
set(LUA_SOURCE_DIR "${CMAKE_SOURCE_DIR}/.deps/lua-5.4.8/src" CACHE PATH "Lua 5.4.8 source")
file(GLOB LUA_SOURCES "${LUA_SOURCE_DIR}/*.c")
list(FILTER LUA_SOURCES EXCLUDE REGEX "/(lua|luac)\\.c$")
add_library(lua STATIC ${LUA_SOURCES})
set_target_properties(lua PROPERTIES ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
# C++ exception unwinding keeps host callback objects safe on Lua allocation/errors.
set_source_files_properties(${LUA_SOURCES} PROPERTIES LANGUAGE CXX)
target_include_directories(lua PUBLIC "${LUA_SOURCE_DIR}")
add_executable(LuaCompiler "${LUA_SOURCE_DIR}/luac.c")
set_source_files_properties("${LUA_SOURCE_DIR}/luac.c" PROPERTIES LANGUAGE CXX)
target_link_libraries(LuaCompiler PRIVATE lua)
set_target_properties(LuaCompiler PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/tools")
file(GLOB_RECURSE EFFECT_SCRIPTS CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/effects/*.lua" "${CMAKE_SOURCE_DIR}/dcs/*.lua")
add_custom_target(CheckLua ALL COMMAND $<TARGET_FILE:LuaCompiler> -p ${EFFECT_SCRIPTS} DEPENDS LuaCompiler VERBATIM)
qt_add_executable(StellaVibra WIN32 src/main.cpp src/model.cpp src/lua_effect.cpp src/engine.cpp src/hid.cpp src/simconnect.cpp src/window.cpp src/window_project.cpp src/window_files.cpp src/window_dialogs.cpp src/window_dcs.cpp src/editor.cpp)
qt_add_resources(StellaVibra app_icon PREFIX "/" FILES assets/stella-vibra.png)
set_target_properties(StellaVibra PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>/bin")
add_executable(StellaLauncher WIN32 src/launcher.cpp)
target_sources(StellaVibra PRIVATE "${CMAKE_BINARY_DIR}/version.rc")
target_sources(StellaLauncher PRIVATE "${CMAKE_BINARY_DIR}/version.rc")
target_compile_definitions(StellaVibra PRIVATE STELLA_VERSION="${PROJECT_VERSION}")
set_target_properties(StellaLauncher PROPERTIES OUTPUT_NAME StellaVibra MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>")
set_target_properties(StellaLauncher PROPERTIES PDB_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/symbols/$<CONFIG>")
target_sources(StellaVibra PRIVATE src/network_telemetry.cpp src/dcs_telemetry.cpp src/dcs_install.cpp)
target_sources(StellaVibra PRIVATE src/diagnostics.cpp)
target_link_libraries(StellaVibra PRIVATE dbghelp shell32 ole32)
target_compile_options(StellaVibra PRIVATE /Zi)
target_link_options(StellaVibra PRIVATE /DEBUG /OPT:REF /OPT:ICF)
target_link_libraries(StellaVibra PRIVATE Qt6::Widgets Qt6::Network lua hid setupapi)
target_compile_definitions(StellaVibra PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(StellaVibra PRIVATE /W4 /utf-8 /permissive-)
if(EXISTS "${CMAKE_SOURCE_DIR}/.deps/msfs2024/client/SimConnect.h")
  target_include_directories(StellaVibra PRIVATE "${CMAKE_SOURCE_DIR}/.deps/msfs2024/client")
  target_compile_definitions(StellaVibra PRIVATE STELLA_CHECK_SIMCONNECT_SDK)
endif()
install(TARGETS StellaLauncher RUNTIME DESTINATION .)
install(TARGETS StellaVibra RUNTIME DESTINATION bin)
install(FILES $<TARGET_PDB_FILE:StellaVibra> DESTINATION bin OPTIONAL)
install(DIRECTORY effects DESTINATION .)
install(DIRECTORY dcs DESTINATION .)
install(FILES LICENSE README.md DESTINATION .)
install(FILES assets/stella-vibra-logo.png DESTINATION assets)
install(FILES docs/MANUAL.md docs/RELEASE-NOTES-0.2.0.md docs/THIRD-PARTY-NOTICES.md DESTINATION docs)
